Dove Eliminare Eventuali Campi da NON Gestire nella Clone di una Pagina

Quando si preme il Tasto di CLONE (al posto del solito Tasto di ADD) il sistema provvede a proporre in ADD un Nuovo Record con tutti i Dati del Record Originario sul quale è stato premuto il Tasto di Clone.

Nel caso si desideri CAMBIARE/ANNULLARE i Valori proposti per ogni singolo Campo Video è possibile farlo intervenendo nella

PowerWeb
    GridPage.Master.cs
         protected void cpCommand_Callback(object sender, CallbackEventArgsBase e)
        //Gestisce il Tasto di Clone -Caricamento Valori dal Record Originale al Record Clonato
        //con eventuale MODIFICA dei Dati Clonati (x adesso solo nella REGV)
        {
            .......................
            if (splitter[0] == "addClone")
            {
                GridClonedValues = new Hashtable();
                //Provvede a Modificare l'EditDictionary nel caso della REGV
                //Eliminando i CAMPI VIDEO che NON si vogliono mostrare nel CLONE
                var templateDic = EditDictionaryManager.GetEditDictionaryRegV();
                foreach (GridViewDataColumn column in grid.Columns.OfType<GridViewDataColumn>())
                //Scorre le Colonne della GRIDVIEW
                {
                    if (grid.KeyFieldName != column.FieldName)
                    //Se la Griglia ha delle Colonne
                    {
                        if (GridModule.EntityType == typeof(Reg_V))
                        //Se si tratta della Griglia REGV
                        {
                            if (column.FieldName == CommonService.GetPropertyName(() => _regvStub.Codice_Fru) ||
                                column.FieldName == CommonService.GetPropertyName(() => _regvStub.Codice_Pru) ||

                                column.FieldName == CommonService.GetPropertyName(() => _regvStub.Registrazione_Tipo_Desc) ||

                                var toDeleteTabs = new List<TabPageExtended>();
                                //Identifica nel EditFormDictiionary Standard le relative definizioni delle Colonne da Eliminare
                                foreach (var keyValuePair in templateDic)
                                {
                                    keyValuePair.Value.RemoveAll(tpi => tpi.Field == column.FieldName);
                                    var emptyFields = keyValuePair.Value.Where(tpi => tpi.Field == null).Count();
                                    if (keyValuePair.Value.Count == 0 || keyValuePair.Value.Count == emptyFields)
                                    //Carica nella Lista delle Definizioni da Cancellare quelle relative alle Colonne interessate
                                        toDeleteTabs.Add(keyValuePair.Key);
                                }
                                //
                                foreach (var R in toDeleteTabs)
                                //Scorre l'Elenco delle Definizioni da Cancellare dall'EditFormDictionary Standard
                                    templateDic.Remove(toDeleteTab);
                                    //Elimina Fisicamente dalla Definizione dell'EditFormDictionary le Definizioni delle Colonne da Cancellare
                                continue;
                            }
                        }

                        GridClonedValues[column.FieldName] = grid.GetRowValues(index, column.FieldName);
                    }
                }
                if (GridModule.EntityType == typeof(Reg_V))
                //Se sto trattando il Clone di un Record della GRID_V sostituisce il Nuovo EditFormDictionary a quello Standard
                    GridView.Templates.EditForm = new PowerFormTemplate(BaseGridModule, templateDic);
                //Carica il Nuovo Record Clonato
                grid.AddNewRow();
            }
