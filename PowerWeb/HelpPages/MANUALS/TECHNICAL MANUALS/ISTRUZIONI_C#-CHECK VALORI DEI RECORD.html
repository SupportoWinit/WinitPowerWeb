CREAZIONE CONTROLLI VALIDITA' CAMPI TABELLA

Tutti i Controlli da fare sui vari campi di una Tabella (sia in INS/VAR da Video, sia durante Import , sia da parte di qualsiasi routine che aggiorna
e/o inserisce un Record in una determinata Tabella) sono

                    SONO TUTTI CENTRALIZZATI NELLA CLASSE CHECK
  •     BUSINESS
    • REPOSITORY
      • CUSTOM
        • es: ICantRepository
          • CantRepository
            •  public override Dictionary<string, string> Check(Tab_Distanze entity, bool isNew = false)
in modo tale che vengano SEMPRE EFFETTUATI da Tutte le Routine che toccano per qualsiasi ragione una Tabella
  • CHECK  x CAMPI OBBLIGATORI
        // Verifico che sia stato immesso un Valore nel campo NomeCampoTabella
              if (String.IsNullOrEmpty(entity.NomeCampoTabella))
                    result.Add(CommonService.GetPropertyName(() => entity.NomeCampoTabella),
                      CommonService.GetLocalizedString(PowerWebResources.ERR_CAMPO_X_E_OBBLIGATORIO, PowerWebResources.FLD_NOMECAMPOTABELLA));
  • CHECK x CAMPI  CHE DEVONO AVERE VALORI MINORI/MAGGIORI DI ALTRI
        // Verifico che il VALORE di un ca mpo NomeCampoTabella1 SIA MINORE del valore di un altro campo NomeCampoTabella2
             if (CommonService.Nz(entity.NomeCampoTabella1, 0) < CommonService.Nz(entity.NomeCampoTabella2, 0))
                    result.Add(CommonService.GetPropertyName(() => entity.NomeCampoTabella1),
                          CommonService.GetLocalizedString(PowerWebResources.ERR_CAMPO_X_DEVE_ESSERE_MINORE_UGUALE_Y,
                              new string[] { PowerWebResources.FLD_NomeCampoTabella1, PowerWebResources.FLD_NomeCampoTabella1 }));
  • CHECK x CAMPI DATA CHE DEVONO AVERE VALORI DATA MINORI/MAGGIORI DI ALTRI
        // Verifico che la DATA di un campo NomeCampoTabella1 sia minore della data di un altro campo NomeCampoTabella2
              if (CommonService.Nz(entity.NomeCampoTabella1, new DateTime(1, 1, 1))
                  > CommonService.Nz(entity.NomeCampoTabella2, new DateTime(9999, 1, 1)))
                    result.Add(CommonService.GetPropertyName(() => entity.NomeCampoTabella1),
                      CommonService.GetLocalizedString(PowerWebResources.ERR_DATA_INIZIO_X_DEVE_ESSERE_MINORE_DI_DATA_FINE_Y,
                          new string[] { PowerWebResources.FLD_NomeCampoTabella1, PowerWebResources.FLD_NomeCampoTabella2 }));

  • CHECK x CAMPI NUMERICI CHE DEVONO AVERE VALORI MINORI DI UN DETERMINATO VALORE
      // Verifico  che il valore del campo NomeCampoTabella1 sia minore di un certo valore predefinito massimo
              if (CommonService.Nz(entity.NomeCampoTabella1, 0) > 31)
                    result.Add(CommonService.GetPropertyName(() => entity.NomeCampoTabella1),
                      CommonService.GetLocalizedString(PowerWebResources.ERR_CAMPO_X_DEVE_ESSERE_MINORE_UGUALE_Y,
                          new string[] { PowerWebResources.FLD_NomeCampoTabella1, "31" }));

  • CHECK x CAMPI CHE DEVONO ESSERE PRESENTI IN UN ALTRA TABELLA NomeTabella
            NomeTabella oNomeTabella = NomeTabellas.SingleOrDefault(x => x.NomeCampoChiaveTabella == oRow.NomeCampoChiaveTabella);

                      oNewRecord.NomeCampoID = oNomeTabella.NomeCampoID;
                else
                    //se NON esiste Segnalo Errore
               
                           
  • CHECK x CAMPI CHE DEVONO ESSERE PRESENTI IN TAB_DECOD (Prendondo o meno ALTRI VALORI ASSOCIATI IN TAB_DECOD)
         // Verifico che il valore del campo NomeCampoTabella1 sia presente nella TAB_DECOD e modifico il valore anche di altri campi
                      if (CommonService.Nz(NomeCampoTabella1, "") != "")  oppure if (CommonService.Nz(entity.NomeCampoTabella1, 0) != 0)
                          {
                              tab_Decod = RepoManager.TabDecodRepo.SearchDecodifyInTable(

                                    CommonService.GetLocalizedString(PowerWebResources.ERR_VALORE_CAMPO_X_NON_PRESENTE_IN_TABELLA_DECODIFICHE,

                                  if (tab_Decod.Associato1 != "")
